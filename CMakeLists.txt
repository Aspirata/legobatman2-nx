cmake_minimum_required(VERSION 3.25)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
project(legobatman2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR NOT MSVC)
  message(FATAL_ERROR "The validated host baseline requires Windows x64 and clang-cl. Horizon is not integrated yet.")
endif()
find_package(rexglue 0.10.0 EXACT REQUIRED CONFIG)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/generated/default/sources.cmake")
  message(FATAL_ERROR "Run python tools/codegen.py before configuring the host.")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/config/game-build.json" GAME_BUILD_JSON)
string(JSON LB2_XEX_SHA256 GET "${GAME_BUILD_JSON}" xex_sha256)
configure_file(config/game_identity.h.in config/game_build.h @ONLY)

add_library(lb2_identity STATIC src/host/game_identity.cpp)
target_include_directories(lb2_identity PUBLIC src/host)
target_link_libraries(lb2_identity PUBLIC bcrypt)

include(generated/default/sources.cmake)
add_library(lb2_guest OBJECT ${GENERATED_SOURCES})
target_include_directories(lb2_guest PRIVATE generated/default)
target_link_libraries(lb2_guest PRIVATE rex::runtime)
target_precompile_headers(lb2_guest PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/generated/default/legobatman2_pch.h")
# Keep CMake's per-configuration optimization (Release /O2, Debug /Od).
# An unconditional /Od makes FMV decoding fall behind the audio clock and
# continuously skip presentation while trying to catch up.
target_compile_options(lb2_guest PRIVATE /EHa /clang:-msse4.1 /clang:-ffp-contract=off /clang:-fno-strict-aliasing)

add_executable(legobatman2 WIN32 src/main.cpp src/host/input_guard.cpp src/host/xma_packet_fix.cpp)
set(LB2_PC_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" CACHE PATH "Playable PC build directory")
set_target_properties(legobatman2 PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${LB2_PC_OUTPUT_DIRECTORY}")
target_include_directories(legobatman2 PRIVATE generated/default "${CMAKE_CURRENT_BINARY_DIR}/config")
target_link_libraries(legobatman2 PRIVATE lb2_guest lb2_identity rex::runtime)
target_compile_options(legobatman2 PRIVATE /clang:-msse4.1 /clang:-ffp-contract=off /clang:-fno-strict-aliasing)
rexglue_configure_target(legobatman2 GPU_PLUGINS xenos)

# App-local Microsoft runtime: no dependency on developer PATH or a system installer.
set(LB2_MSVC_RUNTIME_DIR "${CMAKE_CURRENT_SOURCE_DIR}/work/toolchain/msvc/Contents/VC/Tools/MSVC/14.44.35207/bin/Hostx64/x64" CACHE PATH "Pinned Microsoft runtime DLL source")
set(LB2_MSVC_RUNTIME_DLLS msvcp140.dll msvcp140_atomic_wait.dll vcruntime140.dll vcruntime140_1.dll)
foreach(dll IN LISTS LB2_MSVC_RUNTIME_DLLS)
  if(NOT EXISTS "${LB2_MSVC_RUNTIME_DIR}/${dll}")
    message(FATAL_ERROR "Missing ${dll}; run tools/bootstrap-host.py")
  endif()
  add_custom_command(TARGET legobatman2 POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${LB2_MSVC_RUNTIME_DIR}/${dll}" "$<TARGET_FILE_DIR:legobatman2>" VERBATIM)
endforeach()
