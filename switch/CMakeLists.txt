set(LB2_PREBUILT_GUEST_MANIFEST "" CACHE FILEPATH "Verified ARM64 guest object cache")
if(LB2_PREBUILT_GUEST_MANIFEST)
  file(READ "${LB2_PREBUILT_GUEST_MANIFEST}" _guest_manifest)
  file(SHA256 "${CMAKE_CXX_COMPILER}" _compiler_sha256)
  file(SHA256 "${CMAKE_TOOLCHAIN_FILE}" _toolchain_sha256)
  foreach(identity compiler_sha256 toolchain_sha256 compile_contract)
    string(JSON _expected GET "${_guest_manifest}" "${identity}")
    if(identity STREQUAL "compile_contract")
      set(_actual "${_guest_compile_contract}")
    else()
      set(_actual "${_${identity}}")
    endif()
    if(NOT _actual STREQUAL _expected)
      message(FATAL_ERROR "Stale ARM64 guest cache ${identity}; use -RecompileGuest")
    endif()
  endforeach()
  foreach(group inputs objects)
    string(JSON _count LENGTH "${_guest_manifest}" ${group})
    math(EXPR _last "${_count}-1")
    foreach(index RANGE 0 ${_last})
      string(JSON _path GET "${_guest_manifest}" ${group} ${index} path)
      string(JSON _expected GET "${_guest_manifest}" ${group} ${index} sha256)
      file(SHA256 "${LB2_ROOT}/${_path}" _actual)
      if(NOT _actual STREQUAL _expected)
        message(FATAL_ERROR "Stale ARM64 guest cache: ${_path}")
      endif()
      if(group STREQUAL "objects")
        list(APPEND _guest_objects "${LB2_ROOT}/${_path}")
      endif()
    endforeach()
  endforeach()
  list(LENGTH GENERATED_SOURCES _expected_count)
  list(LENGTH _guest_objects _actual_count)
  if(NOT _expected_count EQUAL _actual_count)
    message(FATAL_ERROR "Incomplete ARM64 guest object cache")
  endif()
  add_library(lb2_switch_guest OBJECT IMPORTED)
  set_property(TARGET lb2_switch_guest PROPERTY IMPORTED_OBJECTS "${_guest_objects}")
else()
